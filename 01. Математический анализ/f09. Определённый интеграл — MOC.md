```dataview
TABLE WITHOUT ID
  file.link AS "Заметка",
  lectures AS "Лекции",
  summary AS "О чём"
FROM "01. Математический анализ"
WHERE up = this.file.link
SORT number(lectures[0]) ASC, file.name ASC
```
